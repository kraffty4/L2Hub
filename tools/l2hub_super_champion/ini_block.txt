# Шанс суперчемпиона, % (дробный: 0.1 — один моб из тысячи; бросается при каждом появлении моба).

# Титул суперчемпиона.
SuperChampionTitle = Legendary Champion

# Множитель HP суперчемпиона.

# Множитель регенерации HP суперчемпиона.
SuperChampionHpRegen = 2.0

# Множитель опыта и SP суперчемпиона.
SuperChampionRewardsExpSp = 20.0

# Множитель шанса обычного дропа суперчемпиона.
SuperChampionRewardsChance = 10.0

# Множитель количества обычного дропа суперчемпиона.
SuperChampionRewardsAmount = 2.0

# Множитель шанса адены и печатей суперчемпиона.
SuperChampionAdenasRewardsChance = 1.0

# Множитель количества адены и печатей суперчемпиона.
SuperChampionAdenasRewardsAmount = 10.0

# Эффекты над моделью (AbnormalVisualEffect через запятую, пусто — без эффектов): BIG_BODY — моб крупнее,

# Примерить на цели в игре: //ave_abnormal <ЭФФЕКТ>.
# Interlude шлёт клиенту только обычные эффекты: особые (INVINCIBILITY и др.) там не видны.
SuperChampionEffects = BIG_BODY,ULTIMATE_DEFENCE,MAGIC_SQUARE

# Объявление при появлении: Region — игрокам в том же регионе карты, World — всем, Off — без объявления.
SuperChampionAnnounce = Region

# Текст объявления: %name% — моб, %level% — его уровень, %town% — ближайший город.
SuperChampionAnnounceText = Легендарный чемпион %name% (%level% ур.) появился у %town%!
