
# ---------------------------------------------------------------------------
# PvP-ивенты Mobius (custom/events: TvT, Deathmatch, CtF) по расписанию — data/scripts/custom/L2HubPvpEvents
# ---------------------------------------------------------------------------
# Для охотников-фантомов: ядро фантомов записывает их на открытую регистрацию (HunterPvpEvents в PhantomHunters.ini).
# Без охотников в одиночку ивент не набирает участников (от 4), и Mobius отменяет его сам.
# False — расписания нет, порог уровня в скриптах ивентов прежний (76, у Deathmatch в H5 и Epilogue — 85).
# Default: False
PvpEvents = False

# Старт только при живом игроке в мире (с клиентом; фантомы и офлайн-торговцы не считаются).
# Default: True
PvpOnlyWithPlayers = True

# Порог уровня участника вместо зашитого в скриптах — действует, пока PvpEvents = True. 1 — любой уровень: на свежем
# мире охотники долго ходят ниже 20-го (прогон Interlude 2026-10-21 — все 7–10-го), и порог 20 не пускал ни одного.
# Default: 1
PvpMinLevel = 1

# Размеры и бой TvT действуют, пока PvpEvents = True, иначе как в Mobius.
# Сколько участников нужно для старта.
# Default: 4
TvTMinParticipants = 4

# Всего мест, до 200: счёт на экране показывает первых 25. Охотников за ивент не больше HunterPvpEventMax в PhantomHunters.ini.
# Default: 24
TvTMaxParticipants = 24

# Сколько мест из TvTMaxParticipants охотники не занимают — они остаются живым игрокам.
# Default: 0
TvTReservedSlots = 0

# Длительность боя, минуты (от 1 до 120).
# Default: 20
TvTFightMinutes = 20

# Расписание — крон «минуты часы день месяц день_недели», как schedule у Mobius; пусто — не запускать.
# Пока идёт один PvP-ивент (менеджер в мире), следующий не стартует. Регистрация 10 мин, бой 20 мин (у TvT — TvTFightMinutes; Deathmatch — 1 и 3).
TvT = 0 */3 * * *
Deathmatch = 0 1-23/3 * * *
CtF = 0 2-23/3 * * *
